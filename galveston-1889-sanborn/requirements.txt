# Environment for the Galveston 1889 Sanborn mosaic pipeline.
#
# Deliberately small, and deliberately free of a system GDAL dependency:
# rasterio ships its own GDAL, and this project uses no gdal_* command-line
# tools and no `osgeo` Python bindings. That removes the single most common
# way these pipelines fail to reproduce, especially on Windows.
#
# Versions below are the ones this pipeline was actually built and tested
# against. Loosen them if you must; these are known to work together.
#
#   python -m venv .venv
#   .venv/bin/pip install -r requirements.txt        # or .venv\Scripts\pip on Windows

numpy==2.4.6
pillow==12.3.0                 # reads the archival JPEG/TIFF scans
opencv-python-headless==5.0.0.93   # warping, correlation, mask rasterising
rasterio==1.4.4                # bundled GDAL 3.10; tiled BigTIFF + windowed IO
affine==3.0.0
shapely==2.1.2                 # seam geometry between mapped regions
pyproj==3.7.2
scipy==1.17.1
pyyaml==6.0.1                  # every human decision lives in YAML
requests==2.33.1               # source acquisition (step 01 only)
beautifulsoup4==4.15.0         # link discovery on the collection page
lxml==6.1.1

# --- used by optional steps only ---
geopandas==1.1.4               # convenience when inspecting GCP/mask layers
scikit-image==0.26.0
matplotlib==3.11.1
tqdm==4.70.0
